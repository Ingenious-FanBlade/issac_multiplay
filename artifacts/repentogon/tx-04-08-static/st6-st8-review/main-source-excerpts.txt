568: 	}

570: 	if (!def->CanHook())

574: 	int argc = def->GetArgCount();

576: 	/// Create the hook caller.
577: 	/// _internalHook will contain the code that will call the hook and place the
578: 	/// arguments on the stack, minus the this pointer, if any (in ecx).
579: 	/// Detours will be used to redirect the original function to the hook caller.
580: 
581: 	unsigned char* ptr = _internalHook;
582: 	ptr = EmitPrologue(def, ptr);
583:

587: 	// so we need to find the "bottom" of the call stack for the active frame.
588: 	uint8_t stackPos = 8;  // Start at 8 because we need to ignore the ebp that was saved in the prologue
589: 							// and we need to ignore the return address below.
590: 	for (int i = 0; i < argc; ++i) {
591: 		if (!args[i].IsRegister()) {

593: 			stackPos += 4 * (uint8_t)args[i]._size;

600: 		}

613: 	// Iterate in reverse order as arguments need to be pushed in reverse order.
614: 	// Farthest argument of the call to the original function must be pushed first.
